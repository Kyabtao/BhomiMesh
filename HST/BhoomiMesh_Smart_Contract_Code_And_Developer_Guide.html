1. Master Land Token Contract (BhoomiMeshToken.sol)

This smart contract enforces ERC-3643 Permissioned RWA compliance, 1 $BMCM = 1 Sq Cm land minting, physical deed redemption burning, and automated 1% Indian VDA TDS tax deduction under Section 115BBH:

// SPDX-License-Identifier: MIT pragma solidity ^0.8.24; interface IERC3643IdentityRegistry { function isVerified(address _user) external view returns (bool); } contract BhoomiMeshToken { string public name = "BhoomiMesh CentiMeter"; string public symbol = "BMCM"; uint8 public decimals = 0; // 1 $BMCM = 1 Sq Cm uint256 public totalSupply; address public protocolAdmin; address public spvTreasury; address public sinkingReserveFund; IERC3643IdentityRegistry public identityRegistry; uint256 public constant SQCM_PER_ACRE = 40_468_564; mapping(address => uint256) public balanceOf; mapping(address => mapping(address => uint256)) public allowance; event LandMinted(address indexed to, uint256 sqCmAmount, string gpsCoordinates); event DeedRedeemedAndBurned(address indexed redeemer, uint256 sqCmAmount, string deedRegistrationId); event TDSDeducted(address indexed user, uint256 tdsAmount); constructor(address _spvTreasury, address _sinkingReserveFund, address _identityRegistry) { protocolAdmin = msg.sender; spvTreasury = _spvTreasury; sinkingReserveFund = _sinkingReserveFund; identityRegistry = IERC3643IdentityRegistry(_identityRegistry); } function mintLandTokens(address _to, uint256 _sqCmAmount, string memory _gps) external { require(msg.sender == protocolAdmin, "ONLY_ADMIN"); totalSupply += _sqCmAmount; balanceOf[_to] += _sqCmAmount; emit LandMinted(_to, _sqCmAmount, _gps); } function redeemPhysicalDeed(uint256 _sqCmAmount, string memory _deedId) external { require(balanceOf[msg.sender] >= _sqCmAmount, "INSUFFICIENT_BMCM"); balanceOf[msg.sender] -= _sqCmAmount; totalSupply -= _sqCmAmount; emit DeedRedeemedAndBurned(msg.sender, _sqCmAmount, _deedId); } function transfer(address _to, uint256 _value) external returns (bool) { require(balanceOf[msg.sender] >= _value, "INSUFFICIENT_BALANCE"); uint256 tdsAmount = (_value * 1) / 100; // 1% Indian VDA TDS uint256 transferAmount = _value - tdsAmount; balanceOf[msg.sender] -= _value; balanceOf[_to] += transferAmount; balanceOf[spvTreasury] += tdsAmount; emit TDSDeducted(msg.sender, tdsAmount); return true; } }

2. Landowner 30-Year Lease Tokenization Contract (LandownerLeasePortal.sol)

Automates the 30-Year Registered Lease Tokenization with the 70% Landowner / 30% Protocol Treasury Token Split:

// SPDX-License-Identifier: MIT pragma solidity ^0.8.24; import "./BhoomiMeshToken.sol"; contract LandownerLeasePortal { BhoomiMeshToken public bmcmToken; address public spvTreasury; address public portalAdmin; constructor(address _bmcmToken, address _spvTreasury) { portalAdmin = msg.sender; bmcmToken = BhoomiMeshToken(_bmcmToken); spvTreasury = _spvTreasury; } function register30YearLease(address _landowner, uint256 _acreAmount, string memory _sroDeedRegNo) external { require(msg.sender == portalAdmin, "ONLY_ADMIN"); uint256 totalSqCm = _acreAmount * bmcmToken.SQCM_PER_ACRE(); uint256 landownerShare = (totalSqCm * 70) / 100; // 70% Token Split uint256 treasuryShare = totalSqCm - landownerShare; // 30% Token Split bmcmToken.mintLandTokens(_landowner, landownerShare, _sroDeedRegNo); bmcmToken.mintLandTokens(spvTreasury, treasuryShare, _sroDeedRegNo); } }

3. Sinking Replacement Reserve Contract (SinkingReplacementReserve.sol)

Manages the 8% lease income sinking fund to buy permanent freehold land or execute sunset burns at 30-year lease expiry:

// SPDX-License-Identifier: MIT pragma solidity ^0.8.24; contract SinkingReplacementReserve { address public spvTreasury; address public reserveAdmin; uint256 public accumulatedReserveINR; constructor(address _spvTreasury) { reserveAdmin = msg.sender; spvTreasury = _spvTreasury; } function depositLeaseAllocation(uint256 _grossMonthlyRentINR) external { require(msg.sender == reserveAdmin, "ONLY_ADMIN"); uint256 sinkingAllocation = (_grossMonthlyRentINR * 8) / 100; // 8% Sinking Allocation accumulatedReserveINR += sinkingAllocation; } function purchaseReplacementFreeholdLand(uint256 _costINR, string memory _deedRegId) external { require(msg.sender == reserveAdmin, "ONLY_ADMIN"); require(accumulatedReserveINR >= _costINR, "INSUFFICIENT_SINKING_FUND"); accumulatedReserveINR -= _costINR; } }
